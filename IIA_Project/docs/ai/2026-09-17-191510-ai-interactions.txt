  Get-NetTCPConnection -State Listen -ErrorAction SilentlyContinue |

  Get-CimInstance Win32_Process -Filter "Name='postgres.exe'" |

  # By port — safest, kills exactly the instance on 5434
  $pids = (Get-NetTCPConnection -LocalPort 5434 -State Listen -ErrorAction
  SilentlyContinue).OwningProcess
  $pids | ForEach-Object { Stop-Process -Id $_ -Force }

  # Or the clean way, if you know the data directory (graceful shutdown, no

  & "C:\Program Files\PostgreSQL\17\bin\pg_ctl.exe" -D "C:\Program

  Get-Process postgres -ErrorAction SilentlyContinue | Stop-Process -Force
